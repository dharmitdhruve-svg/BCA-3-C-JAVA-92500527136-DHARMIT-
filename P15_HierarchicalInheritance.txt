/*
 * Practical 15 : Write a java program to use Hierarchical inheritance
 * Save this file as : HierarchicalInheritance.java
 * Compile : javac HierarchicalInheritance.java
 * Run     : java HierarchicalInheritance
 */

// One parent class
class Vehicle {
    void start() {
        System.out.println("Vehicle is starting...");
    }
}

// Child 1 of Vehicle
class Car extends Vehicle {
    void drive() {
        System.out.println("Car is driving on four wheels.");
    }
}

// Child 2 of Vehicle
class Bike extends Vehicle {
    void ride() {
        System.out.println("Bike is riding on two wheels.");
    }
}

// Child 3 of Vehicle
class Truck extends Vehicle {
    void carryLoad() {
        System.out.println("Truck is carrying heavy load.");
    }
}

public class HierarchicalInheritance {
    public static void main(String[] args) {
        Car c = new Car();
        c.start();          // inherited from Vehicle
        c.drive();

        Bike b = new Bike();
        b.start();          // inherited from Vehicle
        b.ride();

        Truck t = new Truck();
        t.start();          // inherited from Vehicle
        t.carryLoad();
    }
}

/* Output:
 * Vehicle is starting...
 * Car is driving on four wheels.
 * Vehicle is starting...
 * Bike is riding on two wheels.
 * Vehicle is starting...
 * Truck is carrying heavy load.
 */
