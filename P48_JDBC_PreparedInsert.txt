/*
 * Practical 48 : Write a JDBC program to Insert Records Into Oracle Table Using Prepared Statement.
 * Save this file as : PreparedInsert.java
 */
/*
 * ---------------- COMMON SETUP FOR ALL JDBC PRACTICALS ----------------
 * 1) Oracle table (run once in SQL*Plus / SQL Developer):
 *      CREATE TABLE student (id NUMBER PRIMARY KEY, name VARCHAR2(50), marks NUMBER);
 * 2) Download the Oracle driver ojdbc jar (e.g. ojdbc8.jar) and keep it in the same folder.
 * 3) Change URL / USER / PASSWORD below as per your Oracle installation.
 *      URL format : jdbc:oracle:thin:@localhost:1521:xe
 * 4) Compile : javac CLASSNAME.java
 *    Run     : java -cp .;ojdbc8.jar CLASSNAME        (Windows)
 *              java -cp .:ojdbc8.jar CLASSNAME        (Linux / Mac)
 * -----------------------------------------------------------------------
 */
import java.sql.*;
import java.util.Scanner;

public class PreparedInsert {
    public static void main(String[] args) {
        String url  = "jdbc:oracle:thin:@localhost:1521:xe";
        String user = "system";
        String pass = "password";

        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
            Connection con = DriverManager.getConnection(url, user, pass);

            // ? are placeholders (parameters) filled in later
            PreparedStatement ps = con.prepareStatement("INSERT INTO student VALUES (?, ?, ?)");

            Scanner sc = new Scanner(System.in);
            System.out.print("Enter id    : ");
            int id = sc.nextInt();
            System.out.print("Enter name  : ");
            String name = sc.next();
            System.out.print("Enter marks : ");
            int marks = sc.nextInt();

            ps.setInt(1, id);               // 1st ? = id
            ps.setString(2, name);          // 2nd ? = name
            ps.setInt(3, marks);            // 3rd ? = marks

            int rows = ps.executeUpdate();
            System.out.println(rows + " record inserted using PreparedStatement.");

            ps.close();
            con.close();
        } catch (Exception e) {
            System.out.println(e);
        }
    }
}
