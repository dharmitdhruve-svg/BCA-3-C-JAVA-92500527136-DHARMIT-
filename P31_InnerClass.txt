/*
 * Practical 31 : Write a java program to use simple inner class in your program
 * Save this file as : InnerClassDemo.java
 * Compile : javac InnerClassDemo.java
 * Run     : java InnerClassDemo
 */
class Outer {
    private String message = "Hello from Outer class";

    // Inner class : defined inside another class (non-static)
    class Inner {
        void display() {
            // inner class can access even PRIVATE members of outer class
            System.out.println("Inner class says: " + message);
        }
    }
}

public class InnerClassDemo {
    public static void main(String[] args) {
        Outer outer = new Outer();                  // first create outer object
        Outer.Inner inner = outer.new Inner();      // then create inner object using outer object
        inner.display();
    }
}

/* Output:
 * Inner class says: Hello from Outer class
 */
