/*
 * Practical 24 : Write a java program to use Finally block in Exception Handling
 * Save this file as : FinallyDemo.java
 * Compile : javac FinallyDemo.java
 * Run     : java FinallyDemo
 */
public class FinallyDemo {
    public static void main(String[] args) {
        try {
            int data = 25 / 0;                  // exception occurs here
            System.out.println(data);
        } catch (ArithmeticException e) {
            System.out.println("Catch block: " + e.getMessage());
        } finally {
            // finally block ALWAYS executes (exception or not) - used for cleanup
            System.out.println("Finally block: always executed.");
        }
        System.out.println("Rest of the code...");
    }
}

/* Output:
 * Catch block: / by zero
 * Finally block: always executed.
 * Rest of the code...
 */
