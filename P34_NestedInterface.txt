/*
 * Practical 34 : Write a java program to use Nested Interface
 * (An interface declared inside a class or inside another interface)
 * Save this file as : NestedInterfaceDemo.java
 * Compile : javac NestedInterfaceDemo.java
 * Run     : java NestedInterfaceDemo
 */
class Account {

    // Nested interface inside a class
    interface Transaction {
        void perform(double amount);
    }
}

// Implement the nested interface using Outer.Inner name
class Deposit implements Account.Transaction {
    public void perform(double amount) {
        System.out.println("Rs. " + amount + " deposited successfully.");
    }
}

public class NestedInterfaceDemo {
    public static void main(String[] args) {
        Account.Transaction t = new Deposit();
        t.perform(5000);
    }
}

/* Output:
 * Rs. 5000.0 deposited successfully.
 */
