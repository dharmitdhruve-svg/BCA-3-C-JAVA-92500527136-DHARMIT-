/*
 * Practical 18 : Write a java program to use Multiple inheritance using interface.
 * (Java does not allow multiple inheritance with classes, but allows it with interfaces)
 * Save this file as : MultipleInheritanceDemo.java
 * Compile : javac MultipleInheritanceDemo.java
 * Run     : java MultipleInheritanceDemo
 */

// First interface
interface Printable {
    void print();
}

// Second interface
interface Showable {
    void show();
}

// One class implements BOTH interfaces (multiple inheritance)
class Report implements Printable, Showable {

    public void print() {                       // from Printable
        System.out.println("Report is being printed.");
    }

    public void show() {                        // from Showable
        System.out.println("Report is being shown on screen.");
    }
}

public class MultipleInheritanceDemo {
    public static void main(String[] args) {
        Report r = new Report();
        r.print();
        r.show();

        // Same object can be referred by either interface type
        Printable p = r;
        p.print();

        Showable s = r;
        s.show();
    }
}

/* Output:
 * Report is being printed.
 * Report is being shown on screen.
 * Report is being printed.
 * Report is being shown on screen.
 */
