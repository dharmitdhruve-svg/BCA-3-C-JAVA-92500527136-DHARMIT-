/*
 * Practical 19 : Write a java program to use method overriding
 * Save this file as : MethodOverridingDemo.java
 * Compile : javac MethodOverridingDemo.java
 * Run     : java MethodOverridingDemo
 */

// Parent class
class Animal {
    void sound() {
        System.out.println("Animal makes a sound.");
    }
}

// Child class overrides the parent's method (same name, same parameters)
class Dog extends Animal {
    @Override                                   // annotation: checks we really override
    void sound() {
        System.out.println("Dog barks: Bhow Bhow!");
    }
}

class Cat extends Animal {
    @Override
    void sound() {
        super.sound();                          // calls parent's version first
        System.out.println("Cat meows: Meow Meow!");
    }
}

public class MethodOverridingDemo {
    public static void main(String[] args) {
        Animal a;                               // parent reference

        a = new Animal();
        a.sound();                              // parent's method

        a = new Dog();                          // runtime polymorphism
        a.sound();                              // Dog's overridden method

        a = new Cat();
        a.sound();                              // Cat's overridden method
    }
}

/* Output:
 * Animal makes a sound.
 * Dog barks: Bhow Bhow!
 * Animal makes a sound.
 * Cat meows: Meow Meow!
 */
