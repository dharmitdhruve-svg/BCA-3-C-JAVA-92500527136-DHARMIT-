/*
 * Practical 11 : Write a java program to use Interface in java
 * Save this file as : InterfaceDemo.java
 * Compile : javac InterfaceDemo.java
 * Run     : java InterfaceDemo
 */

// An interface contains only abstract methods (by default public & abstract)
interface Shape {
    double PI = 3.14159;          // variables in interface are public static final
    void area();                  // abstract method (no body)
}

// Circle class implements the Shape interface, so it MUST define area()
class Circle implements Shape {
    double radius;

    Circle(double radius) {       // constructor to set radius
        this.radius = radius;
    }

    public void area() {          // implementation of interface method
        System.out.println("Area of Circle = " + (PI * radius * radius));
    }
}

// Rectangle class also implements Shape
class Rectangle implements Shape {
    double length, width;

    Rectangle(double length, double width) {
        this.length = length;
        this.width = width;
    }

    public void area() {
        System.out.println("Area of Rectangle = " + (length * width));
    }
}

// Main class
public class InterfaceDemo {
    public static void main(String[] args) {
        Shape s1 = new Circle(5);          // interface reference -> Circle object
        Shape s2 = new Rectangle(4, 6);    // interface reference -> Rectangle object

        s1.area();                         // calls Circle's area()
        s2.area();                         // calls Rectangle's area()
    }
}

/* Output:
 * Area of Circle = 78.53975
 * Area of Rectangle = 24.0
 */
