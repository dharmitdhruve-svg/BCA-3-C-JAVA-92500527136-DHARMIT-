/*
 * Practical 37 : Write a java program to add, subtract a days/month into current date and time
 * Save this file as : AddSubtractDate.java
 * Compile : javac AddSubtractDate.java
 * Run     : java AddSubtractDate
 */
import java.util.Calendar;
import java.text.SimpleDateFormat;

public class AddSubtractDate {
    public static void main(String[] args) {
        SimpleDateFormat sdf = new SimpleDateFormat("dd-MMM-yyyy HH:mm:ss");
        Calendar cal = Calendar.getInstance();

        System.out.println("Current date and time : " + sdf.format(cal.getTime()));

        cal.add(Calendar.DAY_OF_MONTH, 10);         // add 10 days
        System.out.println("After adding 10 days  : " + sdf.format(cal.getTime()));

        cal.add(Calendar.MONTH, 2);                 // add 2 months
        System.out.println("After adding 2 months : " + sdf.format(cal.getTime()));

        cal.add(Calendar.DAY_OF_MONTH, -5);         // negative value = subtract 5 days
        System.out.println("After subtract 5 days : " + sdf.format(cal.getTime()));

        cal.add(Calendar.MONTH, -1);                // subtract 1 month
        System.out.println("After subtract 1 month: " + sdf.format(cal.getTime()));

        cal.add(Calendar.HOUR, 3);                  // add 3 hours
        System.out.println("After adding 3 hours  : " + sdf.format(cal.getTime()));
    }
}

/* Output: depends on the current date and time */
