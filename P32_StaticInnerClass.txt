/*
 * Practical 32 : Write a java program to use Static Inner Class
 * Save this file as : StaticInnerDemo.java
 * Compile : javac StaticInnerDemo.java
 * Run     : java StaticInnerDemo
 */
class Company {
    static String companyName = "Marwadi Tech";   // static member of outer class
    int employees = 100;                           // non-static member

    // Static nested class : can be created WITHOUT an outer class object
    static class Department {
        void show() {
            // can access only STATIC members of the outer class directly
            System.out.println("Department of " + companyName);
        }
    }
}

public class StaticInnerDemo {
    public static void main(String[] args) {
        // no need to create Company object
        Company.Department d = new Company.Department();
        d.show();
    }
}

/* Output:
 * Department of Marwadi Tech
 */
