/*
 * Practical 27 : Write a java program to use Throws Keyword
 * Save this file as : ThrowsDemo.java
 * Compile : javac ThrowsDemo.java
 * Run     : java ThrowsDemo
 */
import java.io.IOException;

public class ThrowsDemo {

    // 'throws' DECLARES that this method may pass a checked exception to its caller
    static void readData() throws IOException {
        System.out.println("Reading data...");
        throw new IOException("Unable to read data (device error)");
    }

    public static void main(String[] args) {
        try {
            readData();                         // caller must handle or declare the exception
        } catch (IOException e) {
            System.out.println("Handled in main: " + e.getMessage());
        }
        System.out.println("Normal flow continues.");
    }
}

/* Output:
 * Reading data...
 * Handled in main: Unable to read data (device error)
 * Normal flow continues.
 */
