/*
 * Practical 23 : Write a java program to implement Arithmetic Exception
 * Save this file as : ArithmeticExceptionDemo.java
 * Compile : javac ArithmeticExceptionDemo.java
 * Run     : java ArithmeticExceptionDemo
 */
public class ArithmeticExceptionDemo {
    public static void main(String[] args) {
        int a = 50, b = 0;
        try {
            int result = a / b;                 // division by zero -> ArithmeticException
            System.out.println("Result = " + result);
        } catch (ArithmeticException e) {
            System.out.println("Error: Cannot divide by zero.");
            System.out.println("Message: " + e.getMessage());
        }
        System.out.println("End of program.");
    }
}

/* Output:
 * Error: Cannot divide by zero.
 * Message: / by zero
 * End of program.
 */
