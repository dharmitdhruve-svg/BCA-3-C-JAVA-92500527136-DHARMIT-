/*
 * Practical 33 : Write a java program to use Local Inner Class
 * (A class defined INSIDE a method; usable only within that method)
 * Save this file as : LocalInnerDemo.java
 * Compile : javac LocalInnerDemo.java
 * Run     : java LocalInnerDemo
 */
class Calculator {
    void calculate() {
        final int a = 20, b = 10;                   // local variables used by inner class

        // Local inner class : visible only inside calculate()
        class Operation {
            void add() {
                System.out.println("Addition       = " + (a + b));
            }
            void subtract() {
                System.out.println("Subtraction    = " + (a - b));
            }
        }

        Operation op = new Operation();             // object must be created inside the method
        op.add();
        op.subtract();
    }
}

public class LocalInnerDemo {
    public static void main(String[] args) {
        Calculator c = new Calculator();
        c.calculate();
    }
}

/* Output:
 * Addition       = 30
 * Subtraction    = 10
 */
